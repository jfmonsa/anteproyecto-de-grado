\subsubsection{Marco Lógico de la Propuesta}

% --- Objective A ---
\newcommand{\objetivoEspecificoAMetodologia}{
  Desarrollo ágil basado en SCRUM y prácticas DevOps; clasificación de cada flujo según su afinidad con el ciclo de vida de la sesión para decidir entre función \textit{serverless} y BFF.
}

\newcommand{\objetivoEspecificoAEntregables}{
  \begin{itemize}[leftmargin=*, topsep=0pt, itemsep=0pt]
    \item Funciones AWS Lambda para la lógica de servicios e integraciones asíncronas.
    \item Capa \textit{Backend-for-Frontend} en Nuxt que encapsula las integraciones externas.
  \end{itemize}
}

\newcommand{\objetivoEspecificoAIndicador}{
  Porcentaje de lógica crítica reubicada fuera del cliente y flujos sensibles instrumentados para auditoría.
}

\newcommand{\objetivoEspecificoAMeta}{
  Lógica crítica de los módulos de procesos y campañas ejecutándose del lado del servidor, sin exposición en el cliente.
}

% --- Objective B ---
\newcommand{\objetivoEspecificoBMetodologia}{
  Refactorización incremental guiada por patrones arquitectónicos (DDD y separación por capas), validada mediante pruebas unitarias.
}

\newcommand{\objetivoEspecificoBEntregables}{
  \begin{itemize}[leftmargin=*, topsep=0pt, itemsep=0pt]
    \item \textit{Frontend} organizado en Nuxt Layers por \textit{bounded context}.
    \item \textit{Backend} en Go con separación \textit{Handler--Service--Repository} y repositorios por interfaz.
  \end{itemize}
}

\newcommand{\objetivoEspecificoBIndicador}{
  Número de dominios/capas aislados de forma independiente y cobertura de pruebas del \textit{service} con \textit{mocks} del repositorio.
}

\newcommand{\objetivoEspecificoBMeta}{
  \textit{Frontend} y \textit{backend} desacoplados que permitan la evolución independiente de cada dominio y la sustitución del motor de persistencia sin tocar la lógica de negocio.
}
% --- Objective C ---
\newcommand{\objetivoEspecificoCMetodologia}{
  Diseño basado en principios de observabilidad y SRE, con infraestructura como código mediante CloudFormation.
}

\newcommand{\objetivoEspecificoCEntregables}{
  \begin{itemize}[leftmargin=*, topsep=0pt, itemsep=0pt]
    \item Esquema de telemetría estructurada JSON e interceptores HTTP en \textit{frontend} y \textit{backend}.
    \item \textit{Dashboards}, alarmas y notificaciones a Telegram aprovisionados como código.
  \end{itemize}
}

\newcommand{\objetivoEspecificoCIndicador}{
  \begin{itemize}[leftmargin=*, topsep=0pt, itemsep=0pt]
    \item Capacidad de correlacionar eventos de extremo a extremo entre componentes heterogéneos.
    \item Subsistema de observabilidad versionado como código.
  \end{itemize}
}

\newcommand{\objetivoEspecificoCMeta}{
  Subsistema de observabilidad operativo que permita diagnosticar fallos y degradaciones en producción correlacionando eventos en flujos reales.
}

% --- Objective D ---
\newcommand{\objetivoEspecificoDMetodologia}{
  Cultura DevOps con automatización del ciclo de vida del desarrollo mediante \textit{hooks} locales e integración continua.
}

\newcommand{\objetivoEspecificoDEntregables}{
  \begin{itemize}[leftmargin=*, topsep=0pt, itemsep=0pt]
    \item \textit{Hooks} de Git con Lefthook (ESLint, Prettier, Commitlint).
    \item Pipeline de CI en GitHub Actions y suite de pruebas de reglas de seguridad de Firestore con Vitest.
  \end{itemize}
}

\newcommand{\objetivoEspecificoDIndicador}{
  \begin{itemize}[leftmargin=*, topsep=0pt, itemsep=0pt]
    \item Validaciones automatizadas ejecutándose en cada \textit{commit} y \textit{pull request}.
    \item Pruebas de reglas de seguridad operativas.
  \end{itemize}
}

\newcommand{\objetivoEspecificoDMeta}{
  \begin{itemize}[leftmargin=*, topsep=0pt, itemsep=0pt]
    \item Prácticas de aseguramiento de calidad automatizadas y operativas.
    \item Prevención de regresiones funcionales y de seguridad sin depender de la disciplina individual.
  \end{itemize}
}
